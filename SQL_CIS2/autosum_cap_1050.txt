// FORM = 49
// CAPITOL = 1050
// Autosumă pe coloane pentru fiecare rând static
// Col.1 = suma celorlalte coloane active (Col.2 ... Col.8)

function sumCap1050_Col1() {
    var rowIds = {
        "01": "117635",
        "02": "117633",
        "03": "117634",
        "04": "117625",
        "05": "117629",
        "06": "117640",
        "07": "117641",
        "08": "117630",
        "09": "117643",
        "10": "117631",
        "11": "117636",
        "12": "117644",
        "13": "117626",
        "14": "117637",
        "15": "117627",
        "16": "117642",
        "17": "117638",
        "18": "117639",
        "19": "117632",
        "20": "117628"
    };

    Object.keys(rowIds).forEach(function (rind) {
        var sum = 0;

        // În capitolul 1050 sunt definite 8 coloane active.
        for (var col = 2; col <= 8; col++) {
            sum += parseInt(
                $("#49_1050_" + rowIds[rind] + "_" + rind + "_" + col).val(),
                10
            ) || 0;
        }

        $("#49_1050_" + rowIds[rind] + "_" + rind + "_1")
            .val(sum !== 0 ? sum : "")
            .prop("readonly", true);
    });
}

function f_Capitol_1050() {
    sumCap1050_Col1();
}

$(document).ready(function () {
    f_Capitol_1050();

    $("input:not([type='button']):not([readonly]):not([disabled])")
        .on("change", f_Capitol_1050);

    $(document).on(
        "change",
        "input:not([type='button']):not([readonly]):not([disabled])",
        f_Capitol_1050
    );
});
