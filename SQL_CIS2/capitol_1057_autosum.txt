// FORM = 49
// CAPITOL = 1057
// Rânduri statice
//
// Autosume:
// Rând.010 = Rând.020 + Rând.060, Col.1-3
// Rând.020 = Rând.030 + Rând.040 + Rând.050, Col.1-3

var cap1057RowIds = {
    "010": "117693",
    "020": "117694",
    "030": "117689",
    "040": "117695",
    "050": "117691",
    "060": "117690"
};

function sumCap1057_Rind020() {
    var sourceRows = ["030", "040", "050"];

    for (var col = 1; col <= 3; col++) {
        var sum = 0;

        for (var i = 0; i < sourceRows.length; i++) {
            var rind = sourceRows[i];

            sum += parseInt(
                $("#49_1057_" + cap1057RowIds[rind] + "_" + rind + "_" + col).val(),
                10
            ) || 0;
        }

        $("#49_1057_" + cap1057RowIds["020"] + "_020_" + col)
            .val(sum !== 0 ? sum : "")
            .prop("readonly", true);
    }
}

function sumCap1057_Rind010() {
    var sourceRows = ["020", "060"];

    for (var col = 1; col <= 3; col++) {
        var sum = 0;

        for (var i = 0; i < sourceRows.length; i++) {
            var rind = sourceRows[i];

            sum += parseInt(
                $("#49_1057_" + cap1057RowIds[rind] + "_" + rind + "_" + col).val(),
                10
            ) || 0;
        }

        $("#49_1057_" + cap1057RowIds["010"] + "_010_" + col)
            .val(sum !== 0 ? sum : "")
            .prop("readonly", true);
    }
}

function f_Capitol_1057() {
    // Mai întâi calculăm 020, deoarece 010 depinde de el.
    sumCap1057_Rind020();
    sumCap1057_Rind010();
}

$(document).ready(function () {
    f_Capitol_1057();

    $("input:not([type='button']):not([readonly]):not([disabled])")
        .on("change", f_Capitol_1057);

    $(document).on(
        "change",
        "input:not([type='button']):not([readonly]):not([disabled])",
        f_Capitol_1057
    );
});
